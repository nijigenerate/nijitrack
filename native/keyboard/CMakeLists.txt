cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(nijitrack_keyboard LANGUAGES CXX)
endif()
option(NIJITRACK_KEYBOARD "Build native global keyboard backend" ON)
add_library(nijitrack-keyboard STATIC common.cpp)
target_compile_features(nijitrack-keyboard PUBLIC cxx_std_20)
target_include_directories(nijitrack-keyboard PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(NOT NIJITRACK_KEYBOARD)
  target_sources(nijitrack-keyboard PRIVATE unavailable.cpp)
elseif(WIN32)
  target_sources(nijitrack-keyboard PRIVATE windows.cpp)
  target_link_libraries(nijitrack-keyboard PRIVATE user32)
elseif(APPLE)
  target_sources(nijitrack-keyboard PRIVATE macos.cpp)
  target_link_libraries(nijitrack-keyboard PRIVATE "-framework ApplicationServices")
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(X11 REQUIRED)
  if(NOT X11_Xtst_LIB)
    message(FATAL_ERROR "XTest/RECORD library (libXtst) is required")
  endif()
  target_sources(nijitrack-keyboard PRIVATE x11.cpp)
  target_include_directories(nijitrack-keyboard PRIVATE ${X11_INCLUDE_DIR})
  target_link_libraries(nijitrack-keyboard PRIVATE ${X11_X11_LIB} ${X11_Xtst_LIB})
else()
  message(FATAL_ERROR "Unsupported keyboard platform")
endif()
add_executable(keyboard-probe probe.cpp)
target_link_libraries(keyboard-probe PRIVATE nijitrack-keyboard)
