cmake_minimum_required(VERSION 3.22)
project(nijitrack_halide_warp LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
file(TO_CMAKE_PATH "${Halide_ROOT}" Halide_ROOT)
find_package(Halide 15.0.1 EXACT REQUIRED COMPONENTS Halide shared)
add_executable(generate-warp warp_generator.cpp "${Halide_ROOT}/share/Halide/tools/GenGen.cpp")
target_link_libraries(generate-warp PRIVATE Halide::Halide)
add_custom_command(TARGET generate-warp POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:Halide::Halide>" "$<TARGET_FILE_DIR:generate-warp>")
set(output "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${output}")
if(WIN32)
  set(target x86-64-windows)
  set(library "${output}/nijitrack_warp.lib")
else()
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
    set(architecture arm-64)
  else()
    set(architecture x86-64)
  endif()
  if(APPLE)
    set(target "${architecture}-osx")
  else()
    set(target "${architecture}-linux")
  endif()
  set(library "${output}/nijitrack_warp.a")
endif()
add_custom_command(OUTPUT "${library}" "${output}/nijitrack_warp.h"
  COMMAND generate-warp -g nijitrack_warp -f nijitrack_warp -o "${output}" -e static_library,h "target=${target}"
  DEPENDS generate-warp VERBATIM)
add_executable(warp-tests warp_tests.cpp "${output}/nijitrack_warp.h")
get_target_property(headers Halide::Halide INTERFACE_INCLUDE_DIRECTORIES)
target_include_directories(warp-tests PRIVATE "${output}" ${headers})
find_package(Threads REQUIRED)
target_link_libraries(warp-tests PRIVATE "${library}" Threads::Threads ${CMAKE_DL_LIBS})
enable_testing()
add_test(NAME arbitrary-angle-crop-and-normalization COMMAND warp-tests)
