cmake_minimum_required(VERSION 3.24)
project(nijitrack_cpp_prototype VERSION 0.1.0 LANGUAGES CXX)
include(cmake/VerifyModels.cmake)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
option(NIJITRACK_REQUIRE_CLANG "Require Clang for this build" OFF)
if(NIJITRACK_REQUIRE_CLANG AND NOT CMAKE_CXX_COMPILER_ID MATCHES "^(Clang|AppleClang)$")
  message(FATAL_ERROR "This build requires Clang; configure a fresh build directory with clang-cl on Windows")
endif()
set(MEDIAPIPE_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/sdk-halide" CACHE PATH "OpenCV-free face/hand MediaPipe SDK")
set(MEDIAPIPE_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/include" CACHE PATH "Matching official headers")
add_library(MediaPipeArchive STATIC IMPORTED)
if(WIN32)
  if(NOT MSVC)
    message(FATAL_ERROR "Windows requires an MSVC-compatible ABI (clang-cl or MSVC) for the upstream SDK")
  endif()
  set(mediapipe_archive "${MEDIAPIPE_SDK_DIR}/mediapipe_tracking.lib")
  set(halide_archive "${MEDIAPIPE_SDK_DIR}/nijitrack_warp.lib")
else()
  set(mediapipe_archive "${MEDIAPIPE_SDK_DIR}/libmediapipe_tracking.a")
  set(halide_archive "${MEDIAPIPE_SDK_DIR}/libnijitrack_warp.a")
endif()
if(NOT EXISTS "${mediapipe_archive}")
  message(FATAL_ERROR "Build the OpenCV-free static tracking SDK first: ${mediapipe_archive}")
endif()
set_target_properties(MediaPipeArchive PROPERTIES IMPORTED_LOCATION "${mediapipe_archive}")
add_library(MediaPipeC INTERFACE)
# Calculator registration objects must survive archive extraction on every OS.
target_link_libraries(MediaPipeC INTERFACE "$<LINK_LIBRARY:WHOLE_ARCHIVE,MediaPipeArchive>" "${halide_archive}")
target_include_directories(MediaPipeC INTERFACE "${MEDIAPIPE_INCLUDE_DIR}")
if(WIN32)
  target_link_libraries(MediaPipeC INTERFACE bcrypt ws2_32 shlwapi shell32 dbghelp advapi32)
else()
  target_link_libraries(MediaPipeC INTERFACE ${CMAKE_DL_LIBS})
  if(EXISTS "${MEDIAPIPE_SDK_DIR}/mediapipe_tracking_linkopts.txt")
    file(STRINGS "${MEDIAPIPE_SDK_DIR}/mediapipe_tracking_linkopts.txt" tracking_linkopts)
    set(framework_pending OFF)
    foreach(flag IN LISTS tracking_linkopts)
      # Keep each framework pair as one CMake link item. A bare framework name
      # would otherwise become -lName, and repeated -framework flags get deduped.
      if(framework_pending)
        target_link_libraries(MediaPipeC INTERFACE "-framework ${flag}")
        set(framework_pending OFF)
      elseif(flag STREQUAL "-framework")
        set(framework_pending ON)
      else()
        target_link_libraries(MediaPipeC INTERFACE "${flag}")
      endif()
    endforeach()
    if(framework_pending)
      message(FATAL_ERROR "Missing framework name in MediaPipe link options")
    endif()
  endif()
endif()
if(WIN32 AND NOT SDL3_DIR AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/toolchain/SDL3-3.4.18/cmake/SDL3Config.cmake")
  set(SDL3_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/toolchain/SDL3-3.4.18/cmake" CACHE PATH "SDL3 development SDK")
endif()
find_package(SDL3 3.2 REQUIRED CONFIG)
if(NOT WIN32 AND DEFINED ENV{SDL3_ROOT})
  set(SDL3_PACKAGE_ROOT "$ENV{SDL3_ROOT}")
endif()
find_package(Threads REQUIRED)
add_library(nijitrack-media INTERFACE)
target_link_libraries(nijitrack-media INTERFACE SDL3::SDL3 Threads::Threads)
add_library(nijitrack-desktop INTERFACE)
if(WIN32)
  target_link_libraries(nijitrack-desktop INTERFACE ws2_32)
endif()
add_executable(nijitrack-cpp src/main.cpp)
if(WIN32)
  # Use UTF-8 argv and narrow file APIs, including MediaPipe's C API paths.
  target_sources(nijitrack-cpp PRIVATE cmake/nijitrack.manifest)
endif()
if(APPLE)
  target_link_options(nijitrack-cpp PRIVATE
    "LINKER:-sectcreate,__TEXT,__info_plist,${CMAKE_CURRENT_SOURCE_DIR}/cmake/Info.plist")
endif()
add_subdirectory(keyboard)
target_compile_features(nijitrack-cpp PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(nijitrack-cpp PRIVATE /W4 /permissive- /utf-8)
  target_compile_definitions(nijitrack-cpp PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
else()
  target_compile_options(nijitrack-cpp PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
target_link_libraries(nijitrack-cpp PRIVATE MediaPipeC nijitrack-keyboard nijitrack-desktop nijitrack-media)
set_target_properties(nijitrack-cpp PROPERTIES OUTPUT_NAME nijitrack)
add_custom_command(TARGET nijitrack-cpp POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory
    "${CMAKE_CURRENT_SOURCE_DIR}/vendor/models" "$<TARGET_FILE_DIR:nijitrack-cpp>/models")
install(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/vendor/models/" DESTINATION share/nijitrack/models)
if(APPLE)
  set(install_rpath "@loader_path/../lib")
else()
  set(install_rpath "$ORIGIN/../lib")
endif()
set_target_properties(nijitrack-cpp PROPERTIES BUILD_RPATH "${MEDIAPIPE_SDK_DIR}" INSTALL_RPATH "${install_rpath}")
install(TARGETS nijitrack-cpp keyboard-probe RUNTIME DESTINATION bin)
# Remove the historical duplicate when installing into an existing prefix.
install(CODE "file(REMOVE
  \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/bin/nijitrack-cpp${CMAKE_EXECUTABLE_SUFFIX}\")")
if(WIN32)
  add_custom_command(TARGET nijitrack-cpp POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:SDL3::SDL3>" "$<TARGET_FILE_DIR:nijitrack-cpp>")
  install(FILES "$<TARGET_FILE:SDL3::SDL3>" DESTINATION bin)
  install(FILES "${SDL3_DIR}/../LICENSE.txt" DESTINATION share/nijitrack-cpp RENAME LICENSE.sdl3)
  # Old /MD packages copied these DLLs beside the executable. The /MT package
  # no longer uses them; remove only known CRT files at this install prefix.
  install(CODE [[
    file(GLOB obsolete_crt LIST_DIRECTORIES false
      "$ENV{DESTDIR}${CMAKE_INSTALL_PREFIX}/bin/msvcp140*.dll"
      "$ENV{DESTDIR}${CMAKE_INSTALL_PREFIX}/bin/vcruntime140*.dll"
      "$ENV{DESTDIR}${CMAKE_INSTALL_PREFIX}/bin/concrt140.dll"
      "$ENV{DESTDIR}${CMAKE_INSTALL_PREFIX}/bin/vccorlib140.dll")
    if(obsolete_crt)
      file(REMOVE ${obsolete_crt})
    endif()
  ]])
elseif(SDL3_PACKAGE_ROOT)
  install(DIRECTORY "${SDL3_PACKAGE_ROOT}/lib/" DESTINATION lib
    FILES_MATCHING PATTERN "libSDL3*.so*" PATTERN "libSDL3*.dylib")
  install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/vendor/sources/SDL3-3.4.18/LICENSE.txt" DESTINATION share/nijitrack-cpp RENAME LICENSE.sdl3)
endif()
install(FILES "${MEDIAPIPE_SDK_DIR}/LICENSE.mediapipe" "${MEDIAPIPE_SDK_DIR}/LICENSE.halide"
  "${MEDIAPIPE_SDK_DIR}/SOURCE.txt" DESTINATION share/nijitrack-cpp)
install(DIRECTORY "${MEDIAPIPE_SDK_DIR}/licenses/" DESTINATION share/nijitrack-cpp/licenses)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../LICENSE" DESTINATION share/nijitrack-cpp RENAME LICENSE.nijitrack)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/vendor/FACE_CONNECTIONS_SOURCE.txt"
  "${CMAKE_CURRENT_SOURCE_DIR}/vendor/face_mesh_connections.py" DESTINATION share/nijitrack-cpp)
include(CTest)
if(BUILD_TESTING)
  add_executable(tracking-sdk-smoke tests/tracking_sdk_smoke.cpp)
  if(WIN32)
    target_sources(tracking-sdk-smoke PRIVATE cmake/nijitrack.manifest)
  endif()
  target_compile_features(tracking-sdk-smoke PRIVATE cxx_std_20)
  target_link_libraries(tracking-sdk-smoke PRIVATE MediaPipeC Threads::Threads)
  add_executable(stream-tests tests/stream_tests.cpp)
  target_include_directories(stream-tests PRIVATE src)
  target_compile_features(stream-tests PRIVATE cxx_std_20)
  target_link_libraries(stream-tests PRIVATE Threads::Threads)
  add_test(NAME threaded-frame-pipeline COMMAND stream-tests)
  set_tests_properties(threaded-frame-pipeline PROPERTIES TIMEOUT 15)
  add_executable(cli-compatibility tests/cli_compatibility.cpp)
  target_compile_features(cli-compatibility PRIVATE cxx_std_20)
  target_link_libraries(cli-compatibility PRIVATE MediaPipeC nijitrack-keyboard nijitrack-desktop nijitrack-media)
  if(WIN32)
    target_compile_definitions(cli-compatibility PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(cli-compatibility PRIVATE /utf-8)
  endif()
  add_test(NAME python-cli-compatibility COMMAND cli-compatibility)
  add_executable(process-smoke tests/process_smoke.cpp)
  target_include_directories(process-smoke PRIVATE src)
  target_compile_features(process-smoke PRIVATE cxx_std_20)
  if(WIN32)
    target_link_libraries(process-smoke PRIVATE ws2_32)
  endif()
  add_test(NAME process-arguments-and-exit COMMAND process-smoke)
  add_test(NAME cli-help COMMAND nijitrack-cpp --help)
  add_test(NAME python-command-help
    COMMAND "$<TARGET_FILE_DIR:nijitrack-cpp>/nijitrack${CMAKE_EXECUTABLE_SUFFIX}" -h)
  set_tests_properties(cli-help PROPERTIES PASS_REGULAR_EXPRESSION "nijitrack --image")
  add_test(NAME cli-missing-value COMMAND nijitrack-cpp --image)
  set_tests_properties(cli-missing-value PROPERTIES WILL_FAIL TRUE)
endif()
