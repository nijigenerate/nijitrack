cmake_minimum_required(VERSION 3.20)
project(nijitrack_cpp_prototype VERSION 0.1.0 LANGUAGES CXX)
set(MEDIAPIPE_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/sdk" CACHE PATH "Pinned tqphan/libmediapipe SDK")
set(MEDIAPIPE_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/include" CACHE PATH "Matching official headers")
add_library(MediaPipeC SHARED IMPORTED)
if(WIN32)
  if(NOT MSVC)
    message(FATAL_ERROR "Windows requires the MSVC toolchain for the upstream SDK")
  endif()
  # The upstream ZIP ships DLLs without an import library. Generate only the
  # linker import table for the official C functions used by this executable.
  set(import_library "${CMAKE_CURRENT_BINARY_DIR}/mediapipe.lib")
  execute_process(COMMAND "${CMAKE_AR}" /nologo /machine:x64
    "/def:${CMAKE_CURRENT_SOURCE_DIR}/cmake/mediapipe.def" "/out:${import_library}"
    RESULT_VARIABLE import_result COMMAND_ERROR_IS_FATAL ANY)
  set_target_properties(MediaPipeC PROPERTIES IMPORTED_LOCATION "${MEDIAPIPE_SDK_DIR}/libmediapipe.dll"
    IMPORTED_IMPLIB "${import_library}")
elseif(APPLE)
  set_target_properties(MediaPipeC PROPERTIES IMPORTED_LOCATION "${MEDIAPIPE_SDK_DIR}/libmediapipe.dylib")
else()
  set_target_properties(MediaPipeC PROPERTIES IMPORTED_LOCATION "${MEDIAPIPE_SDK_DIR}/libmediapipe.so")
endif()
set_target_properties(MediaPipeC PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${MEDIAPIPE_INCLUDE_DIR}")
if(WIN32 AND NOT OpenCV_DIR AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/opencv/build/OpenCVConfig.cmake")
  set(OpenCV_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/opencv/build" CACHE PATH "Matching OpenCV development package")
endif()
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc videoio highgui)
find_package(Threads REQUIRED)
add_library(nijitrack-opencv INTERFACE)
target_include_directories(nijitrack-opencv INTERFACE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(nijitrack-opencv INTERFACE ${OpenCV_LIBS} Threads::Threads)
if(APPLE)
  enable_language(OBJCXX)
  add_library(nijitrack-desktop STATIC src/desktop_macos.mm)
  set_target_properties(nijitrack-desktop PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED YES)
  target_compile_options(nijitrack-desktop PRIVATE -fobjc-arc -Wall -Wextra)
  target_link_libraries(nijitrack-desktop PRIVATE MediaPipeC
    "-framework AVFoundation")
else()
  add_library(nijitrack-desktop INTERFACE)
  if(WIN32)
    target_link_libraries(nijitrack-desktop INTERFACE ws2_32 ole32 oleaut32 strmiids)
  endif()
endif()
add_executable(nijitrack-cpp src/main.cpp)
if(APPLE)
  target_link_options(nijitrack-cpp PRIVATE
    "LINKER:-sectcreate,__TEXT,__info_plist,${CMAKE_CURRENT_SOURCE_DIR}/cmake/Info.plist")
endif()
add_subdirectory(keyboard)
target_compile_features(nijitrack-cpp PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(nijitrack-cpp PRIVATE /W4 /permissive- /utf-8)
  target_compile_definitions(nijitrack-cpp PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
else()
  target_compile_options(nijitrack-cpp PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
target_link_libraries(nijitrack-cpp PRIVATE MediaPipeC nijitrack-keyboard nijitrack-desktop nijitrack-opencv)
add_custom_command(TARGET nijitrack-cpp POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:nijitrack-cpp>"
    "$<TARGET_FILE_DIR:nijitrack-cpp>/nijitrack${CMAKE_EXECUTABLE_SUFFIX}")
if(APPLE)
  set(install_rpath "@loader_path/../lib")
else()
  set(install_rpath "$ORIGIN/../lib")
endif()
set_target_properties(nijitrack-cpp PROPERTIES BUILD_RPATH "${MEDIAPIPE_SDK_DIR}" INSTALL_RPATH "${install_rpath}")
install(TARGETS nijitrack-cpp keyboard-probe RUNTIME DESTINATION bin)
# Copy the installed executable after CMake rewrites its RPATH.
install(CODE "execute_process(COMMAND \"${CMAKE_COMMAND}\" -E copy_if_different
  \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/bin/nijitrack-cpp${CMAKE_EXECUTABLE_SUFFIX}\"
  \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/bin/nijitrack${CMAKE_EXECUTABLE_SUFFIX}\"
  COMMAND_ERROR_IS_FATAL ANY)")
if(WIN32)
  file(GLOB sdk_libraries "${MEDIAPIPE_SDK_DIR}/*.dll")
  # OpenCV's video decoder is a DLL loaded in-process, not ffmpeg.exe.
  file(GLOB video_plugins "${OpenCV_CONFIG_PATH}/../bin/opencv_videoio_*.dll")
  add_custom_command(TARGET nijitrack-cpp POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different ${sdk_libraries} ${video_plugins} "$<TARGET_FILE_DIR:nijitrack-cpp>")
  install(FILES ${sdk_libraries} ${video_plugins} DESTINATION bin)
  if(EXISTS "${OpenCV_INSTALL_PATH}/../sources/3rdparty/ffmpeg/license.txt")
    install(FILES "${OpenCV_INSTALL_PATH}/../sources/3rdparty/ffmpeg/license.txt"
      DESTINATION share/nijitrack-cpp RENAME LICENSE.opencv-videoio-ffmpeg)
  endif()
elseif(APPLE)
  install(FILES "${MEDIAPIPE_SDK_DIR}/libmediapipe.dylib" DESTINATION lib)
else()
  install(FILES "${MEDIAPIPE_SDK_DIR}/libmediapipe.so" DESTINATION lib)
endif()
install(FILES "${MEDIAPIPE_SDK_DIR}/LICENSE.mediapipe" "${MEDIAPIPE_SDK_DIR}/LICENSE.opencv"
  "${MEDIAPIPE_SDK_DIR}/SOURCE.txt" DESTINATION share/nijitrack-cpp)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../LICENSE" DESTINATION share/nijitrack-cpp RENAME LICENSE.nijitrack)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/vendor/FACE_CONNECTIONS_SOURCE.txt"
  "${CMAKE_CURRENT_SOURCE_DIR}/vendor/face_mesh_connections.py" DESTINATION share/nijitrack-cpp)
include(CTest)
if(BUILD_TESTING)
  add_executable(stream-tests tests/stream_tests.cpp)
  target_include_directories(stream-tests PRIVATE src)
  target_compile_features(stream-tests PRIVATE cxx_std_20)
  target_link_libraries(stream-tests PRIVATE Threads::Threads)
  add_test(NAME threaded-frame-pipeline COMMAND stream-tests)
  set_tests_properties(threaded-frame-pipeline PROPERTIES TIMEOUT 15)
  add_executable(cli-compatibility tests/cli_compatibility.cpp)
  target_compile_features(cli-compatibility PRIVATE cxx_std_20)
  target_link_libraries(cli-compatibility PRIVATE MediaPipeC nijitrack-keyboard nijitrack-desktop nijitrack-opencv)
  if(WIN32)
    target_compile_definitions(cli-compatibility PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(cli-compatibility PRIVATE /utf-8)
  endif()
  add_test(NAME python-cli-compatibility COMMAND cli-compatibility)
  add_executable(process-smoke tests/process_smoke.cpp)
  target_include_directories(process-smoke PRIVATE src)
  target_compile_features(process-smoke PRIVATE cxx_std_20)
  if(WIN32)
    target_link_libraries(process-smoke PRIVATE ws2_32)
  endif()
  add_test(NAME process-arguments-and-exit COMMAND process-smoke)
  add_test(NAME cli-help COMMAND nijitrack-cpp --help)
  add_test(NAME python-command-help
    COMMAND "$<TARGET_FILE_DIR:nijitrack-cpp>/nijitrack${CMAKE_EXECUTABLE_SUFFIX}" -h)
  set_tests_properties(cli-help PROPERTIES PASS_REGULAR_EXPRESSION "nijitrack-cpp --image")
  add_test(NAME cli-missing-value COMMAND nijitrack-cpp --image)
  set_tests_properties(cli-missing-value PROPERTIES WILL_FAIL TRUE)
endif()
